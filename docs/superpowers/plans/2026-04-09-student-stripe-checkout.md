# TAT-94: Student Stripe Checkout Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Enable students to pay subscription fees via Stripe Checkout (hosted page) and allow managers to register Pix keys for receiving transfers.

**Architecture:** Adds a `POST /v1/payments/checkout` endpoint that creates a Stripe Checkout Session (mode: payment), returns the URL to the mobile app which opens it in an in-app browser. A webhook handler records the payment on completion. Manager Pix key is stored on the `academies` table with CRUD via `PUT/DELETE /v1/academies/pix-key`.

**Tech Stack:** Stripe Checkout Sessions, Express, Drizzle ORM, PostgreSQL, Expo WebBrowser, React Query, React Email/Resend

async createCheckoutSession(
  studentUserId: string,
  academyId: string,
  stripe: Stripe,
  opts?: { subscriptionId?: string },
): Promise<{ checkoutUrl: string }> {
  // Find active subscription
  const subs = await this.studentSubRepo.listWithPlanByUserAndAcademy(
    studentUserId,
    academyId,
  );
  const active = opts?.subscriptionId
    ? subs.find(
        (s) =>
          s.subscription.id === opts.subscriptionId &&
          s.subscription.status === "active",
      )
    : subs.find((s) => s.subscription.status === "active");

  if (!active) {
    throw new HttpError(404, "No active subscription found");
  }

  const { subscription, plan } = active;

  // Check not already paid this month
  const today = new Date().toISOString().slice(0, 10);
  const referenceMonth = today.slice(0, 7) + "-01";

  const alreadyPaid = await this.paymentRepo.existsForSubscriptionReferenceMonth(
    subscription.id,
    referenceMonth,
  );
  if (alreadyPaid) {
    throw new HttpError(409, "Payment already exists for this period");
  }

  // Create Checkout Session
  const session = await stripe.checkout.sessions.create({
    mode: "payment",
    line_items: [
      {
        price_data: {
          currency: "brl",
          unit_amount: plan.priceCents,
          product_data: { name: `${plan.name} — ${referenceMonth.slice(0, 7)}` },
        },
        quantity: 1,
      },
    ],
    metadata: {
      subscriptionId: subscription.id,
      userId: studentUserId,
      academyId,
      referenceMonth,
    },
    success_url: `tatame://payment-success?session_id={CHECKOUT_SESSION_ID}`,
    cancel_url: "tatame://payment-cancel",
  });

  if (!session.url) {
    throw new HttpError(500, "Failed to create checkout session");
  }

  return { checkoutUrl: session.url };
}
```

private async onCheckoutSessionCompleted(
  session: Stripe.Checkout.Session,
): Promise<void> {
  const meta = session.metadata;
  if (!meta?.subscriptionId || !meta.userId || !meta.academyId || !meta.referenceMonth) {
    return; // Not a student payment checkout
  }

  if (!this.paymentRepo) return;

  // Idempotency: check if payment already exists
  const paymentIntentId =
    typeof session.payment_intent === "string"
      ? session.payment_intent
      : session.payment_intent?.id;

  if (paymentIntentId) {
    const existing = await this.paymentRepo.findByStripePaymentId(paymentIntentId);
    if (existing) return;
  }

  const payment = await this.paymentRepo.create({
    subscriptionId: meta.subscriptionId,
    userId: meta.userId,
    academyId: meta.academyId,
    amountCents: session.amount_total ?? 0,
    method: "stripe",
    confirmedBy: null,
    stripePaymentId: paymentIntentId ?? session.id,
    referenceMonth: meta.referenceMonth,
    notes: null,
    paidAt: new Date(),
  });

  if (!this.emailService) return;

  // Send confirmation email to student
  const student = await this.userRepo.findById(meta.userId);
  const academy = await this.academyRepo.findById(meta.academyId);
  if (!student || !academy) return;

  await this.emailService.sendPaymentConfirmed(
    {
      email: student.email,
      studentName: student.name,
      academyName: academy.name,
      amountCents: payment.amountCents,
      referenceMonth: payment.referenceMonth,
    },
    payment.id,
  );

  // Send notification email to manager
  const owner = await this.userRepo.findById(academy.ownerId);
  if (!owner) return;

  await this.emailService.sendStudentPaymentReceived(
    {
      email: owner.email,
      managerName: owner.name,
      academyName: academy.name,
      studentName: student.name,
      amountCents: payment.amountCents,
      referenceMonth: payment.referenceMonth,
      hasPixKey: !!academy.pixKey,
    },
    payment.id,
  );
}
```
